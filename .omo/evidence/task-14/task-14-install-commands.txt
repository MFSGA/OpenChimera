# Helper: run a step, capture output and exit code
run_step() {
    local step_name="$1"
    shift
    echo ""
    echo "--- Step: $step_name ---"
    "$@"
    echo "exit: $?"
}

# Helper: assert a condition
assert_pass() {
    local description="$1"
    if "$@"; then
        echo "  [PASS] $description"
    else
        echo "  [FAIL] $description"
    fi
}

# Helper: assert that a command returns expected exit code
assert_exit() {
    local expected="$1"
    local description="$2"
    shift 2
    "$@"
    local actual=$?
    if [ "$actual" = "$expected" ]; then
        echo "  [PASS] $description (exit $actual)"
    else
        echo "  [FAIL] $description (expected exit $expected, got $actual)"
    fi
}

# Helper: assert that pgrep returns NO process
assert_no_process() {
    local pattern="${1:-mihomo|chimera}"
    if pgrep -f "$pattern" >/dev/null 2>&1; then
        echo "  [FAIL] Process found running: $(pgrep -f "$pattern" | head -3)"
        return 1
    else
        echo "  [PASS] No process matching '$pattern'"
        return 0
    fi
}

# Helper: assert that pgrep returns a process
assert_has_process() {
    local pattern="${1:-mihomo|chimera}"
    local pid
    pid=$(pgrep -f "$pattern" | head -1)
    if [ -n "$pid" ]; then
        echo "  [PASS] Process running with PID=$pid"
        return 0
    else
        echo "  [FAIL] No process found matching '$pattern'"
        return 1
    fi
}
